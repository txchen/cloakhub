import { createHash, randomUUID } from 'node:crypto';
import { hostname } from 'node:os';
import { basename, dirname, isAbsolute, join } from 'node:path';
import { lstat, mkdir, open, readFile, realpath, rename, rmdir, unlink } from 'node:fs/promises';

const object = value => value !== null && typeof value === 'object' && !Array.isArray(value);
const text = value => typeof value === 'string' && value.trim().length > 0;
function fail(code, message) { const error = new Error(`Checkpoint ${code}: ${message}`); error.code = code; throw error; }
function canonical(value) {
  if (value === null || typeof value === 'boolean' || typeof value === 'string') return JSON.stringify(value);
  if (typeof value === 'number' && Number.isFinite(value)) return JSON.stringify(value);
  if (Array.isArray(value)) return '[' + value.map(canonical).join(',') + ']';
  if (object(value) && Object.getPrototypeOf(value) === Object.prototype) {
    return '{' + Object.keys(value).sort().map(key => JSON.stringify(key) + ':' + canonical(value[key])).join(',') + '}';
  }
  fail('INVALID', 'Use finite JSON data only');
}
const equal = (a, b) => canonical(a) === canonical(b);
const copy = value => JSON.parse(canonical(value));
function origin(value, label) {
  let url; try { url = new URL(value); } catch { fail('INVALID', `${label} must be an exact HTTP(S) origin; correct the adapter field before retrying`); }
  if (!['http:', 'https:'].includes(url.protocol) || url.origin !== value || url.username || url.password) fail('INVALID', `${label} must be an exact origin without paths or credentials`);
  return value;
}
function destination(value, label = 'destination') {
  if (!object(value) || !text(value.accountKey)) fail('INVALID', `${label} must contain siteOrigin and a freshly observed accountKey`);
  return { siteOrigin: origin(value.siteOrigin, `${label}.siteOrigin`), accountKey: value.accountKey };
}
async function regular(path) {
  const stat = await lstat(path);
  if (!stat.isFile() || stat.nlink !== 1) fail('STORAGE', 'Checkpoint must be a regular, unlinked file');
  return stat;
}
async function atomicWrite(path, value) {
  const temp = `${path}.${randomUUID()}.tmp`;
  let file;
  try {
    file = await open(temp, 'wx', 0o600);
    await file.writeFile(JSON.stringify(value) + '\n');
    await file.sync(); await file.close(); file = null;
    await rename(temp, path);
    // Make the rename durable, not just the contents of the temporary file.
    const dir = await open(dirname(path), 'r');
    try { await dir.sync(); } finally { await dir.close(); }
  } finally {
    if (file) await file.close();
    await unlink(temp).catch(error => { if (error.code !== 'ENOENT') throw error; });
  }
}
function alive(pid) {
  try { process.kill(pid, 0); return true; }
  catch (error) { if (error.code === 'ESRCH') return false; return true; }
}
async function lock(path) {
  const dir = path + '.lock';
  const owner = { pid: process.pid, host: hostname(), nonce: randomUUID() };
  for (let attempt = 0; attempt < 3; attempt++) {
    try {
      await mkdir(dir, { mode: 0o700 });
      const f = await open(join(dir, 'owner.json'), 'wx', 0o600);
      try { await f.writeFile(JSON.stringify(owner)); await f.sync(); } finally { await f.close(); }
      return async () => {
        const actual = JSON.parse(await readFile(join(dir, 'owner.json'), 'utf8'));
        if (!equal(actual, owner)) fail('LOCKED', 'Lock ownership changed');
        await unlink(join(dir, 'owner.json')); await rmdir(dir);
      };
    } catch (error) { if (error.code !== 'EEXIST') throw error; }
    let old;
    try {
      if (!(await lstat(dir)).isDirectory() || (await lstat(dir)).isSymbolicLink()) throw Error();
      old = JSON.parse(await readFile(join(dir, 'owner.json'), 'utf8'));
    } catch { fail('LOCKED', 'Incomplete lock; inspect the owning process before manual recovery'); }
    if (old.host !== hostname() || !Number.isSafeInteger(old.pid) || old.pid <= 0 || !text(old.nonce) || alive(old.pid)) {
      fail('LOCKED', 'Another writer may still own this checkpoint');
    }
    // Only one reaper may remove this dead owner's directory. Missing/partial locks fail closed.
    const reap = join(dir, 'reaping');
    try { await mkdir(reap); } catch { fail('LOCKED', 'Checkpoint lock recovery is already in progress'); }
    try {
      const actual = JSON.parse(await readFile(join(dir, 'owner.json'), 'utf8'));
      if (!equal(actual, old) || alive(old.pid)) fail('LOCKED', 'Lock ownership changed during recovery');
      await unlink(join(dir, 'owner.json'));
    } finally { await rmdir(reap); }
    await rmdir(dir);
  }
  fail('LOCKED', 'Could not acquire checkpoint');
}
function validateCheckpoint(saved, identity, rows) {
  if (!object(saved) || saved.version !== 1 || !equal(saved.identity, identity)) fail('IDENTITY', 'Input, operation or destination differs from checkpoint');
  if (!Number.isSafeInteger(saved.revision) || saved.revision < 1 || !Array.isArray(saved.completed) || !['active', 'complete'].includes(saved.status)) fail('CORRUPT', 'Invalid checkpoint state');
  const keys = new Set(rows.map(row => row.key)), seen = new Set(), ids = new Set();
  for (const item of saved.completed) {
    if (!object(item) || !keys.has(item.key) || seen.has(item.key) || !text(item.recordId) || ids.has(item.recordId)) fail('CORRUPT', 'Invalid confirmed records');
    seen.add(item.key); ids.add(item.recordId);
  }
  if (saved.pending !== null && (!object(saved.pending) || !keys.has(saved.pending.key) || seen.has(saved.pending.key) || !text(saved.pending.attemptId) || !text(saved.pending.startedAt))) fail('CORRUPT', 'Invalid pending record');
  if (saved.status === 'complete' && (saved.pending !== null || seen.size !== rows.length)) fail('CORRUPT', 'Invalid completion state');
}

/** The callbacks supply site evidence and UI actions; this function owns the durable state machine. */
export async function checkpointBatch(options) {
  const { path: requestedPath, operation, target, read, submit, onProgress, maxItems = 10, budgetMs = 120000 } = options;
  if (!isAbsolute(requestedPath || '') || !text(operation) || typeof read !== 'function' || typeof submit !== 'function' || (onProgress !== undefined && typeof onProgress !== 'function')) fail('INVALID', 'Need absolute path, operation, rows, destination, read and submit');
  if (!Number.isSafeInteger(maxItems) || maxItems < 1 || maxItems > 100 || !Number.isSafeInteger(budgetMs) || budgetMs < 1) fail('INVALID', 'Invalid batch bounds');
  const expectedDestination = destination(options.destination);
  if (!object(target) || !text(target.profile)) fail('INVALID', 'Exact connected target required');
  const rows = copy(options.rows);
  if (!Array.isArray(rows) || rows.some(row => !object(row) || !text(row.key) || Object.hasOwn(row, 'recordId'))) fail('INVALID', 'Rows need nonempty string keys; recordId is reserved');
  if (new Set(rows.map(row => row.key)).size !== rows.length) fail('INVALID', 'Duplicate input business keys');
  const identity = { operation, hubOrigin: origin(target.url, 'connected hub origin'), profileId: target.profile, ...expectedDestination,
    inputDigest: createHash('sha256').update(canonical(rows)).digest('hex') };
  const path = join(await realpath(dirname(requestedPath)), basename(requestedPath));
  const release = await lock(path);
  const started = Date.now();
  let saved;
  try {
    try { await regular(path); saved = JSON.parse(await readFile(path, 'utf8')); validateCheckpoint(saved, identity, rows); }
    catch (error) {
      if (error.code !== 'ENOENT') throw error;
      saved = { version: 1, identity, revision: 0, status: 'active', completed: [], pending: null };
    }
    const persist = async () => { saved.revision++; await atomicWrite(path, saved); };
    if (saved.revision === 0) await persist();
    const intended = new Map(rows.map(row => [row.key, row]));
    const reconcile = async () => {
      const snapshot = await read();
      if (!object(snapshot) || !equal(destination(snapshot.destination, 'read().destination'), expectedDestination)) fail('IDENTITY', 'Fresh site account/origin differs');
      if (snapshot.complete !== true || !Array.isArray(snapshot.records)) fail('INCOMPLETE', 'Wait for complete relevant records before writing');
      const found = new Map(), ids = new Set();
      for (const record of snapshot.records) {
        if (!object(record) || !text(record.key) || !text(record.recordId)) fail('EVIDENCE', 'Saved records need separate key and durable recordId fields');
        if (!intended.has(record.key)) continue;
        if (found.has(record.key) || ids.has(record.recordId)) fail('DUPLICATE', 'Multiple matching records; stop for reconciliation');
        const { recordId, ...fields } = record;
        if (!equal(fields, intended.get(record.key))) fail('MISMATCH', 'Saved fields differ from intended input');
        found.set(record.key, recordId); ids.add(recordId);
      }
      for (const item of saved.completed) if (found.get(item.key) !== item.recordId) fail('MISSING_CONFIRMED', 'A previously confirmed record changed or disappeared');
      if (saved.pending !== null && !found.has(saved.pending.key)) fail('PENDING', 'Previous submission outcome is unknown; no automatic replay');
      saved.completed = rows.filter(row => found.has(row.key)).map(row => ({ key: row.key, recordId: found.get(row.key) }));
      saved.pending = null;
      saved.status = saved.completed.length === rows.length ? 'complete' : 'active';
      await persist();
      return found;
    };
    let confirmed = await reconcile(), processed = 0;
    for (const row of rows) {
      if (confirmed.has(row.key)) continue;
      if (processed >= maxItems || Date.now() - started >= budgetMs) break;
      // Recheck the account, entire relevant set and known completions immediately before each write.
      if (processed > 0) confirmed = await reconcile();
      if (confirmed.has(row.key)) continue;
      saved.pending = { key: row.key, attemptId: randomUUID(), startedAt: new Date().toISOString() };
      saved.status = 'active';
      await persist(); // A failure here prevents the site write.
      await submit(copy(row)); // Any thrown value leaves the pending record durable.
      confirmed = await reconcile(); // Missing, duplicate or wrong fields stop all further writes.
      processed++;
      if (onProgress) await onProgress({ completed: saved.completed.length, total: rows.length, processed });
    }
    return { path, status: saved.status, completed: saved.completed.length, total: rows.length, processed, pending: saved.pending?.key ?? null, revision: saved.revision };
  } finally { await release(); }
}
